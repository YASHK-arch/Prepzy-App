Q3. Design Library Management System

Problem Description

Design a Library Borrowing System where members borrow books, librarians issue and return books, a book cannot be borrowed by multiple members at the same time, premium members receive fine discounts and all outputs are stored in a shared global log.

Class Definitions

1. Base Class: Member

  • name : Member's name (String)
  • memberId : Member's ID (String)
  • booksBorrowed : Books borrowed (int)
  • totalBorrows : Global borrow count (static int)
  • log : Shared output log (ArrayList<String>)

Behavior:

  • borrowBook(librarian, days)
  • returnBook(librarian, lateDays)
  • applyFineDiscount(fine)

2. Child Class: PremiumMember (extends Member)

  • discountPercent : Fine discount percentage
  • Overrides applyFineDiscount()

3. Class: Librarian

  • name : Librarian’s name (String)
  • employeeId : Unique employee ID
  • booksHandled : Total issue + return operations
  • totalFineCollected : Total fine collected
  • bookCopy : Nested BookCopy object

4. Nested Class: BookCopy (inside Librarian)

  • title : Book title
  • finePerDay : Fine per late day
  • isIssued : Book issued status

Example Walkthrough

Borrow Conflict Example

Member m1 = new Member("Amit", "M101"); Member m2 = new Member("Neha", "M102"); Librarian lib = new Librarian("Rohit", "L201", "JavaBook", 5); m1.borrowBook(lib, 7); m2.borrowBook(lib, 5);
Output (stored in log):
Book borrowed by Amit via Librarian Rohit
Librarian Rohit issued book to Amit for 7 days
Book already issued. Neha cannot borrow

Instructions

  • Implement Member, PremiumMember, Librarian, and nested BookCopy exactly.
  • All output messages must be stored in Member.log only.
  • Do not print directly inside classes.
  • Fine is calculated only if lateDays > 0.
      
// 1. Base Class: Member
class Member {
    String name;
    String memberId;
    int booksBorrowed;

    static int totalBorrows = 0;
    static ArrayList log = new ArrayList<>();

    Member(String name, String memberId) {
        this.name = name;
        this.memberId = memberId;
        this.booksBorrowed = 0;
    }

    void borrowBook(Librarian librarian, int days) {
        if (librarian.bookCopy.isIssued) {
            Member.log.add("Book already issued. " + this.name + " cannot borrow");
            return;
        } else {
            booksBorrowed++;
            Member.totalBorrows++;
            Member.log.add("Book borrowed by " + this.name + " via Librarian " + librarian.name);
            librarian.issueBook(this, days);
        }
    }

    void returnBook(Librarian librarian, int lateDays) {
        if (this.booksBorrowed == 0) return;
        booksBorrowed--;
        librarian.returnBook(this, lateDays);
    }

    double applyFineDiscount(double fine) {
        return fine;
    }
}

// 2. Child Class: PremiumMember
class PremiumMember extends Member {
    double discountPercent;

    PremiumMember(String name, String memberId, double discountPercent) {
        super(name, memberId);
        this.discountPercent = discountPercent;
    }

    @Override
    double applyFineDiscount(double fine) {
        return fine - (fine * discountPercent / 100);
    }
}

// 3. Class: Librarian
class Librarian {
    String name;
    String employeeId;
    int booksHandled;
    double totalFineCollected;
    BookCopy bookCopy;

    Librarian(String name, String employeeId, String title, double finePerDay) {
        this.name = name;
        this.employeeId = employeeId;
        this.booksHandled = 0;
        this.totalFineCollected = 0;
        this.bookCopy = new BookCopy(title, finePerDay);

        Member.log.add("Librarian added: " + this.name + " | Book=" + title + " | FinePerDay=" + finePerDay);
    }

    void issueBook(Member member, int days) {
        if (bookCopy.isIssued) {
            Member.log.add("Book already issued. " + member.name + " cannot borrow");
        } else {
            bookCopy.isIssued = true;
            booksHandled++;
            Member.log.add("Librarian " + this.name + " issued book to " + member.name + " for " + days + " days");
        }
    }

    void returnBook(Member member, int lateDays) {
        double fine = bookCopy.calculateFine(lateDays);
        double finalFine = member.applyFineDiscount(fine);
        totalFineCollected += finalFine;
        booksHandled++;
        bookCopy.isIssued = false;
        Member.log.add("Book returned by " + member.name + " | Fine: " + finalFine);
    }

    // 4. Nested Class: BookCopy
    class BookCopy {
        String title;
        double finePerDay;
        boolean isIssued;

        BookCopy(String title, double finePerDay) {
            this.title = title;
            this.finePerDay = finePerDay;
            this.isIssued = false;
        }

        double calculateFine(int lateDays) {
            if (lateDays <= 0) return 0;
            return lateDays * finePerDay;
        }
    }
}
