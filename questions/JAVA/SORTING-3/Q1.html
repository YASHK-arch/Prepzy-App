Q2. Inversion Count in an Array

Problem Description

Given an array of integers A. If i < j and A[i] > A[j], then the pair (i, j) is called an inversion of A.

Find the total number of inversions of A modulo (109 + 7).

Problem Constraints

  • 1 ≤ length of the array ≤ 105
  • 1 ≤ A[i] ≤ 109

Input Format

The only argument given is the integer array A.

Output Format

Return the number of inversions of A modulo (109 + 7).

Example Input

A = [1, 3, 2]

Example Output

1

Example Explanation

The pair (1, 2) is an inversion since index 1 < 2 and A[1] > A[2]. 

      
import java.util.*;

public class Solution {

    public int solve(int[] A) {
        long modulo = 1000000007L;
        long result = countInversions(A, 0, A.length - 1) % modulo;
        return (int) result;
    }

    // Recursively divide the array and count inversions
    private long countInversions(int[] arr, int l, int r) {
        long count = 0;

        if (l < r) {
            int m = (l + r) / 2;

            // Count inversions in left half
            count += countInversions(arr, l, m);

            // Count inversions in right half
            count += countInversions(arr, m + 1, r);

            // Count cross inversions
            count += mergeAndCount(arr, l, m, r);
        }

        return count;
    }

    // Merge two sorted halves and count inversions
    private long mergeAndCount(int[] arr, int l, int m, int r) {

        int[] left = Arrays.copyOfRange(arr, l, m + 1);
        int[] right = Arrays.copyOfRange(arr, m + 1, r + 1);

        int i = 0, j = 0, k = l;
        long swaps = 0;

        while (i < left.length && j < right.length) {
            if (left[i] <= right[j]) {
                arr[k++] = left[i++];
            } else {
                arr[k++] = right[j++];
                swaps += (left.length - i);
            }
        }

        while (i < left.length) {
            arr[k++] = left[i++];
        }

        while (j < right.length) {
            arr[k++] = right[j++];
        }

        return swaps;
    }
}
